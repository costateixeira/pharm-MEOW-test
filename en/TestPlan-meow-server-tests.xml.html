<TestPlan xmlns="http://hl7.org/fhir"
          resourceDefinition="http://hl7.org/fhir/StructureDefinition/TestPlan|0.1.0-snapshot1">

  <url
       value="http://example.com/fhir/ihe.pharm.meow.test/TestPlan/meow-server-tests"/>
  <version value="1.0.0"/>
  <name value="MEOWServerTestPlan"/>
  <title value="MEOW Responder (Server) Test Plan"/>
  <status value="draft"/>
  <experimental value="true"/>
  <date value="2026-08-26"/>
  <publisher value="Jose Costa Teixeira"/>
  <contact>
    <name value="Jose Costa Teixeira"/>
    <telecom>
      <system value="url"/>
      <value value="https://github.com/costateixeira/pharm-MEOW-test"/>
    </telecom>
  </contact>
  <description
               value="Test plan for the **Medication Overview Responder** actor (the *MEOW server*) defined by the
[IHE PHARM Medication Overview (MEOW)](https://profiles.ihe.net/PHARM/MEOW/index.html) implementation guide.

The suites below are split one per transaction - PHARM-11 Query Medication Resources and PHARM-12
Retrieve Medication Overview Document - matching the one-transaction-per-scenario structure of the
Gherkin script. Each `suite.test` corresponds to one `Scenario:` in that script, matched by its
`tc-meow-server-NNN` identifier.

The system under test is the **server**: the test bed originates every request and asserts on the
response it gets back."/>
  <jurisdiction>
    <coding>
      <system value="http://unstats.un.org/unsd/methods/m49/m49.htm"/>
      <code value="001"/>
      <display value="World"/>
    </coding>
  </jurisdiction>
  <purpose
           value="To declare, in a machine-readable and runnable form, which behaviours a system claiming conformance to the MEOW Medication Overview Responder actor must demonstrate."/>
  <scope>
    <reference
               value="https://profiles.ihe.net/PHARM/MEOW/CapabilityStatement/MedicationOverviewResponder"/>
    <description
                 value="The Medication Overview Responder (MEOW server) actor - the system under test."/>
  </scope>
  <runner value="https://www.itb.ec.europa.eu/docs/guides/latest/"/>
  <mode>
    <code value="required"/>
    <description
                 value="Tests every Medication Overview Responder must pass."/>
  </mode>
  <mode>
    <code value="document-option"/>
    <description
                 value="Tests that only apply to a Responder claiming the Document Option (PHARM-12)."/>
  </mode>
  <suite>
    <name value="PHARM-11 Query Medication Resources"/>
    <description
                 value="Search on MedicationStatement, profiled as MedicationTreatmentLine. The patient parameter is required; status, effective, category and _lastUpdated are optional."/>
    <mode value="required"/>
    <input>
      <name value="gherkin-script"/>
      <file value="meow-server-gherkin-script.feature"/>
    </input>
    <test>
      <name value="tc-meow-server-001 PHARM-11 query by patient"/>
      <description
                   value="The required baseline: a patient-scoped search returns that patient's treatment lines."/>
      <operation value="gherkin/Scenario"/>
      <assertion>
        <severity value="error"/>
        <human
               value="The response is 200 and a Bundle of type searchset with at least one entry."/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="Every entry is a MedicationStatement whose subject resolves to the requested patient."/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="Every entry declares the MedicationTreatmentLine profile in meta.profile."/>
      </assertion>
    </test>
    <test>
      <name
            value="tc-meow-server-002 PHARM-11 honours the optional search parameters"/>
      <description
                   value="status, category and _lastUpdated actually filter server-side rather than being ignored."/>
      <operation value="gherkin/Scenario"/>
      <assertion>
        <severity value="error"/>
        <human
               value="A search filtered by status=active returns 200 and every entry carries that status."/>
      </assertion>
      <assertion>
        <severity value="warning"/>
        <human
               value="A search filtered by category returns 200 and a searchset."/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="A search with _lastUpdated set in the future returns 200 and an empty searchset - proof the parameter is applied rather than ignored."/>
      </assertion>
    </test>
    <test>
      <name
            value="tc-meow-server-003 PHARM-11 rejects a query without the required patient"/>
      <description
                   value="Negative test: an unscoped MedicationStatement search must be refused, not answered."/>
      <operation value="gherkin/Scenario"/>
      <assertion>
        <severity value="error"/>
        <human value="The response status is 400."/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="The body is an OperationOutcome carrying at least one issue of severity error or fatal."/>
      </assertion>
    </test>
  </suite>
  <suite>
    <name value="PHARM-12 Retrieve Medication Overview Document"/>
    <description
                 value="Document Option. Search for and read a Medication Overview document Bundle, then validate it against the MedicationOverview profile."/>
    <mode value="document-option"/>
    <input>
      <name value="gherkin-script"/>
      <file value="meow-server-gherkin-script.feature"/>
    </input>
    <test>
      <name
            value="tc-meow-server-004 PHARM-12 returns a conformant MedicationOverview"/>
      <description
                   value="A document search returns document Bundles, and a document read returns one that validates against the profile."/>
      <operation value="gherkin/Scenario"/>
      <assertion>
        <severity value="error"/>
        <human
               value="A search with patient and type=document returns 200 and a searchset whose entries are all Bundles of type document."/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="Reading a Medication Overview document returns 200 and a Bundle of type document."/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="The retrieved document validates against the MEOW MedicationOverview profile."/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="The document holds exactly one Composition, plus a Patient and at least one MedicationStatement."/>
      </assertion>
    </test>
  </suite>
</TestPlan>
